var DEAD_DAYS = [2, 5, 7, 14];
function parseDue(dead, di) {
  if (di !== undefined && di !== null && DEAD_DAYS[di]) return yd(DEAD_DAYS[di]);
  var s = String(dead || ''), m;
  if ((m = s.match(/(20\d\d)-(\d{1,2})-(\d{1,2})/))) return m[1] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[3]).slice(-2);
  if ((m = s.match(/(\d{1,2})[.\/](\d{1,2})(?:[.\/](\d{2,4}))?/))) {
    var dd = Number(m[1]), mm = Number(m[2]); if (dd < 1 || dd > 31 || mm < 1 || mm > 12) return null;
    var y = m[3] ? Number(m[3].length == 2 ? '20' + m[3] : m[3]) : Number(yd(0).slice(0, 4));
    var r = y + '-' + ('0' + mm).slice(-2) + '-' + ('0' + dd).slice(-2);
    if (!m[3] && r < yd(0)) r = (y + 1) + r.slice(4);
    return r;
  }
  return null;
}

function sha(s) { return Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8)); }
function randStr(n) {
  var abc = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789', out = '';
  while (out.length < n) {
    var hex = Utilities.getUuid().replace(/-/g, '');
    for (var i = 0; i + 2 <= hex.length && out.length < n; i += 2) { var v = parseInt(hex.substr(i, 2), 16); if (v < 232) out += abc[v % abc.length]; }
  }
  return out;
}

function crmRoute(r) {
  var c = CacheService.getScriptCache(), ver = P.getProperty('crm_ver') || '0';
  if (r.a == 'login') {
    var fails = Number(c.get('crm_fail') || 0);
    if (fails >= 8) return {ok: false, err: 'locked'};
    var salt = P.getProperty('crm_salt'), h = P.getProperty('crm_hash');
    if (!h || String(r.u || '').trim() !== P.getProperty('crm_user') || sha(salt + String(r.p || '')) !== h) {
      c.put('crm_fail', String(fails + 1), 900);
      if (fails + 1 == 8 && admin()) call('sendMessage', {chat_id: admin(), text: '⚠️ CRM: 8 սխալ մուտքի փորձ։ Մուտքը փակված է 15 րոպեով։'});
      Utilities.sleep(700);
      return {ok: false, err: 'bad'};
    }
    var tok = randStr(40); c.put('crmt_' + ver + '_' + tok, '1', 21600); c.remove('crm_fail');
    return {ok: true, token: tok};
  }
  var key = 'crmt_' + ver + '_' + String(r.t || '');
  if (!r.t || !c.get(key)) return {ok: false, err: 'auth'};
  c.put(key, '1', 21600);

  if (r.a == 'data') return crmData();
  if (r.a == 'logout') { c.remove(key); return {ok: true}; }

  if (r.a == 'upd') {
    var id = String(r.id), o = getJ('o_' + id, null); if (!o) return {ok: false, err: 'nf'};
    var sent = false;
    if (r.sum !== undefined) o.sum = r.sum === '' || r.sum === null ? '' : Number(r.sum);
    if (r.due !== undefined) { if (r.due != o.due) o.pre = false; o.due = r.due || ''; }
    if (r.note !== undefined) o.note = String(r.note).slice(0, 2000);
    if (r.topic !== undefined) o.topic = String(r.topic).slice(0, 500);
    if (r.status && ST_CODES.indexOf(r.status) >= 0 && r.status != 'done') {
      var changed = o.status != r.status; o.status = r.status;
      if (changed && r.notify && o.uid && TX.hy.st[r.status]) {
        var cu = U(o.uid), m = call('sendMessage', {chat_id: o.uid, parse_mode: 'HTML', text: L(cu).st[r.status].replace('{id}', id)});
        if (m.ok) { markOut(o.uid); sent = true; }
      }
    }
    setJ('o_' + id, o); crm(id);
    return {ok: true, sent: sent};
  }
  if (r.a == 'done') {
    var did = String(r.id), d = getJ('o_' + did, null); if (!d) return {ok: false, err: 'nf'};
    var first = d.status != 'done'; d.status = 'done'; setJ('o_' + did, d); crm(did);
    if (first) { var dc = Number(P.getProperty('doneCount') || 0) + 1; P.setProperty('doneCount', String(dc)); updateCounter(dc); }
    var asked = d.uid ? askReview(did) : false;
    return {ok: true, asked: !!asked};
  }
  if (r.a == 'add') {
    var nid = Number(P.getProperty('next') || 1); P.setProperty('next', String(nid + 1));
    setJ('o_' + nid, {uid: 0, manual: true, client: String(r.client || '').slice(0, 100), type: r.type || 'other', topic: String(r.topic || '').slice(0, 500), dead: r.due || '', due: r.due || '', time: ytime(), ts: now(), status: 'new', lang: r.lang || 'hy', src: String(r.src || 'manual').slice(0, 40), meth: false, sum: r.sum ? Number(r.sum) : '', note: String(r.note || '').slice(0, 2000)});
    crm(nid);
    return {ok: true, id: nid};
  }
  if (r.a == 'msg') {
    var uid = String(r.uid || ''); if (!uid || uid == '0') return {ok: false, err: 'nouid'};
    var text;
    if (r.tpl) {
      var tu = U(uid), q = L2(tu).q[r.tpl]; if (!q) return {ok: false, err: 'tpl'};
      if ((r.tpl == 'price' || r.tpl == 'prepay') && !(Number(r.x) > 0)) return {ok: false, err: 'sum'};
      text = q.replace('{id}', r.oid || tu.order || '—').replace('{x}', fmtNum(r.x || '')).replace('{d}', REVISION_DAYS);
      if (r.tpl == 'price' && r.oid) { var po = getJ('o_' + r.oid, null); if (po) { po.sum = Number(r.x); setJ('o_' + r.oid, po); crm(r.oid); } }
    } else text = esc(String(r.text || '').slice(0, 3500));
    if (!text) return {ok: false, err: 'empty'};
    var s = call('sendMessage', {chat_id: uid, parse_mode: 'HTML', text: text});
    if (s.ok) markOut(uid);
    return {ok: !!s.ok, err: s.ok ? null : 'tg'};
  }
  if (r.a == 'ban') {

function crmData() {
  var all = P.getProperties(), orders = [], users = [], reviews = [];
  Object.keys(all).forEach(function (k) {
    var v; try { v = k.indexOf('o_') == 0 || k.indexOf('u_') == 0 || k.indexOf('rv_') == 0 ? JSON.parse(all[k]) : null; } catch (e) { v = null; }
    if (!v) return;
    if (k.indexOf('o_') == 0) { v.id = Number(k.slice(2)); orders.push(v); }
    else if (k.indexOf('u_') == 0) users.push({id: k.slice(2), name: v.name || '', lang: v.lang || '', src: v.src || 'direct', seen: v.seen || '', subs: v.subs || 0, started: v.started || 0, state: v.state || '', lastIn: v.lastIn || 0, lastOut: v.lastOut || 0, order: v.order || null, banned: all['ban_' + k.slice(2)] == '1'});
    else { v.id = Number(k.slice(3)); reviews.push(v); }
  });
  var names = {}; users.forEach(function (u) { names[u.id] = u.name; });
  orders.forEach(function (o) { o.client = o.client || names[String(o.uid)] || ''; });
  var types = {}; TYPEKEYS.forEach(function (k) { types[k] = tname(k, 'ru'); });
  return {ok: true, now: now(), today: yd(0), orders: orders, users: users, reviews: reviews, types: types,
    done: Number(all.doneCount || 0), sheet: all.sheet ? 'https://docs.google.com/spreadsheets/d/' + all.sheet + '/edit' : '',
    queue: {idx: Number(all.postIdx || 0), total: POSTS.length}, revDays: REVISION_DAYS};
}
