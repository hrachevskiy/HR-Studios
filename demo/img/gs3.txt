function statsText() {
  var keys = P.getKeys(), users = 0, started = 0, subs = 0, langs = {}, src = {}, steps = {type: 0, topic: 0, dead: 0, meth: 0, details: 0};
  keys.forEach(function (k) {
    if (k.indexOf('u_') != 0) return; var u = getJ(k, null); if (!u) return; users++;
    langs[u.lang || '—'] = (langs[u.lang || '—'] || 0) + 1;
    var s = u.src || 'direct'; src[s] = src[s] || {u: 0, o: 0}; src[s].u++; if (u.subs) src[s].o++;
    if (u.started) started++; if (u.subs) subs++;
    if (steps[u.state] !== undefined) steps[u.state]++;
  });
  var done = Number(P.getProperty('doneCount') || 0), orders = Number(P.getProperty('next') || 1) - 1;
  var lines = ['📈 <b>Վիճակագրություն</b>', '', '👥 Օգտատերեր՝ <b>' + users + '</b>', '🛒 Սկսել են պատվեր՝ <b>' + started + '</b>', '✅ Ուղարկել են պատվեր՝ <b>' + subs + '</b> (' + (started ? Math.round(100 * subs / started) : 0) + '%)', '📦 Պատվերներ ընդհանուր՝ ' + orders + ' · ավարտված՝ ' + done, '',
    '⏸ Կանգնած են քայլում՝ տեսակ ' + steps.type + ' · թեմա ' + steps.topic + ' · ժամկետ ' + steps.dead + ' · մեթոդական ' + steps.meth + ' · ֆայլեր ' + steps.details, '',
    '🌐 Լեզուներ՝ ' + Object.keys(langs).map(function (k) { return k + ' ' + langs[k]; }).join(' · '), '', '🔗 <b>Աղբյուրներ</b> (օգտատեր / պատվեր)'];

}

function handleMsg(m) {
  if (m.chat.type != 'private') return;
  var chat = m.chat.id, uid = m.from.id, t = m.text || '';
  if (t.indexOf('/admin') == 0) {
    var fails = Number(P.getProperty('af_' + uid) || 0);
    if (fails >= 5) return;
    if (t.split(/\s+/).pop() == ADMIN_CODE) { P.setProperty('admin', String(chat)); call('sendMessage', {chat_id: chat, text: '✅ Դու ադմին ես։ Բոլոր պատվերներն ու հաղորդագրությունները կգան այստեղ։ /help'}); }
    else P.setProperty('af_' + uid, String(fails + 1));
    return;
  }
  if (admin() && String(chat) == admin()) return handleAdmin(m);
  var u = U(uid);
  if (t == '/start' || t == '/menu' || t.indexOf('/start ') == 0) {
    var arg = t.split(' ')[1] || '';

    if (arg.indexOf('src_') == 0) arg = '';
    if (arg == 'ru' || arg == 'en' || arg == 'hy') { u.lang = arg; saveU(uid, u); return showMenu(chat, u); }
    if (!u.lang) { u.pendingStart = arg; saveU(uid, u); return call('sendMessage', {chat_id: chat, text: '🌐 Ընտրիր լեզուն · Выберите язык · Choose language', reply_markup: langMenu()}); }
    if (arg == 'order') return startOrder(chat, uid);
    return showMenu(chat, u);
  }
  var tt = L(u);
  if (t == '/lang') return call('sendMessage', {chat_id: chat, text: '🌐 Ընտրիր լեզուն · Выберите язык · Choose language', reply_markup: langMenu()});
  if (t == '/order') return startOrder(chat, uid);
  if (t == '/free') return sendFree(chat, u);
  if (t == '/guarantees') return call('sendMessage', {chat_id: chat, parse_mode: 'HTML', text: tt.guar.replace('{d}', REVISION_DAYS) + tt.foot, reply_markup: kb([[btn(tt.b_order_s, 'order')]])});

  if (t == '/help') return call('sendMessage', {chat_id: chat, text: tt.how + tt.foot, parse_mode: 'HTML', reply_markup: kb([[btn(tt.b_order_s, 'order')]])});
  var st = u.state;
  if (st == 'rtext') { u.rev = u.rev || {}; u.rev.text = t || ''; u.state = 'rpub'; saveU(uid, u); return call('sendMessage', {chat_id: chat, parse_mode: 'HTML', text: tt.rpub, reply_markup: kb([[btn(tt.b_yes, 'rp_1'), btn(tt.b_no, 'rp_0')]])}); }
  if ((st == 'topic' || st == 'dead') && !t) return call('sendMessage', {chat_id: chat, text: tt.textonly});
  if (st == 'topic') { u.draft.topic = t; u.state = 'dead'; touch(u); saveU(uid, u); return askDead(chat, u); }
  if (st == 'dead') { u.draft.dead = t; u.state = 'meth'; touch(u); saveU(uid, u); return askMeth(chat, u); }
  if (st == 'meth') {
    u.draft.msgs = u.draft.msgs || []; u.draft.msgs.push(m.message_id); u.draft.meth = true; touch(u); saveU(uid, u);
    if (u.draft.msgs.length == 1) call('sendMessage', {chat_id: chat, text: '📥', reply_markup: kb([[btn(tt.b_methdone, 'meth_skip')]])});

  relayToAdmin(m, uid);

// every 30 min: remind users who started an order and stopped (>= 2 h)
function remindAbandoned() {
  var keys = P.getKeys(), lim = now() - REMIND_AFTER_MIN * 60000;
  keys.forEach(function (k) {
    if (k.indexOf('u_') != 0) return;
    var u = getJ(k, null); if (!u || u.rem || !u.ts) return;
    if (['type', 'topic', 'dead', 'details'].indexOf(u.state) < 0) return;
    if (u.ts > lim) return;
    var t = L(u), uid = k.slice(2);
    call('sendMessage', {chat_id: uid, text: t.remind, reply_markup: kb([[btn(t.b_cont, 'resume')], [btn(t.b_cancel, 'cancel')]])});
    u.rem = true; setJ(k, u);
  });
}
// every hour: on Mon/Wed/Fri at 19:00 Yerevan publish next post from POSTS
function postNext(force) {
  var d = new Date(), day = Number(Utilities.formatDate(d, 'Asia/Yerevan', 'u')), hr = Number(Utilities.formatDate(d, 'Asia/Yerevan', 'H'));
  var today = Utilities.formatDate(d, 'Asia/Yerevan', 'yyyy-MM-dd');
  if (!force) { if ([1, 3, 5].indexOf(day) < 0 || hr != 19 || P.getProperty('lastPost') == today) return 'skip'; }
  var idx = Number(P.getProperty('postIdx') || 0);
  if (idx >= POSTS.length) { if (admin() && P.getProperty('qEmpty') != '1') { call('sendMessage', {chat_id: admin(), text: '🗓 Ալիքի գրառումների հերթը վերջացավ։ Գրիր, որ նորերը ավելացնենք։'}); P.setProperty('qEmpty', '1'); } return 'empty'; }
  var p = POSTS[idx], r;
  var markup = kb([[ubtn(p.btn || '🤖 Պատվիրել բոտում', BOT_LINK + '?start=' + (p.start || 'order'))]]);
  if (p.video) r = call('sendVideo', {chat_id: CHAN_ID, video: SITE + p.video, caption: p.text, parse_mode: 'HTML', supports_streaming: true, reply_markup: markup});
